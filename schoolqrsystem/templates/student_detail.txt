    <script>
        // Ambil student_id dari URL
        const urlParams = new URLSearchParams(window.location.search);
        const studentId = urlParams.get('id');
        
        // Load data saat halaman dimuat
        document.addEventListener('DOMContentLoaded', function() {
            if (studentId) {
                loadStudentData(studentId);
                loadAttendanceHistory(studentId);
            } else {
                document.body.innerHTML = '<h1>ID Siswa tidak ditemukan</h1>';
            }
        });
        
        // Fungsi untuk memuat data siswa
        async function loadStudentData(id) {
            try {
                const response = await fetch(`/student/${id}`);
                const data = await response.json();
                
                if (data.success) {
                    const student = data.student;
                    
                    // Update informasi siswa
                    document.getElementById('studentName').textContent = student.name;
                    
                    // Update status badge
                    const statusBadge = document.getElementById('statusBadge');
                    statusBadge.textContent = student.is_active ? 'AKTIF' : 'TIDAK AKTIF';
                    statusBadge.className = `status-badge ${student.is_active ? 'status-present' : 'status-absent'}`;
                    
                    // Update informasi detail
                    document.getElementById('studentInfo').innerHTML = `
                        <div class="info-item">
                            <div class="info-label">ID Siswa</div>
                            <div class="info-value">${student.student_id}</div>
                        </div>
                        <div class="info-item">
                            <div class="info-label">Nama Lengkap</div>
                            <div class="info-value">${student.name}</div>
                        </div>
                        <div class="info-item">
                            <div class="info-label">Kelas</div>
                            <div class="info-value">${student.class_name}</div>
                        </div>
                        <div class="info-item">
                            <div class="info-label">Tanggal Daftar</div>
                            <div class="info-value">${new Date(student.created_at).toLocaleDateString('id-ID', {
                                weekday: 'long',
                                year: 'numeric',
                                month: 'long',
                                day: 'numeric'
                            })}</div>
                        </div>
                        <div class="info-item">
                            <div class="info-label">Total Presensi</div>
                            <div class="info-value">${student.total_attendance} kali</div>
                        </div>
                    `;
                    
                    // Update QR code
                    const qrCodeImage = document.getElementById('qrCodeImage');
                    qrCodeImage.src = `/qrcode-view/${id}`;
                    
                    // Update download link
                    document.getElementById('downloadQR').href = `/qrcode/${id}`;
                }
            } catch (error) {
                console.error('Error:', error);
            }
        }
        
        // Fungsi untuk memuat riwayat presensi
        async function loadAttendanceHistory(id) {
            try {
                const response = await fetch(`/student/${id}`);
                const data = await response.json();
                
                if (data.success && data.student.attendance_history.length > 0) {
                    const tableBody = document.getElementById('attendanceTableBody');
                    const noAttendanceDiv = document.getElementById('noAttendance');
                    
                    tableBody.innerHTML = '';
                    document.getElementById('attendanceTable').style.display = 'table';
                    noAttendanceDiv.style.display = 'none';
                    
                    data.student.attendance_history.forEach(record => {
                        const row = document.createElement('tr');
                        
                        const statusClass = record.status === 'present' ? 'status-present' : 
                                          record.status === 'late' ? 'status-late' : 'status-absent';
                        
                        row.innerHTML = `
                            <td>${record.timestamp}</td>
                            <td>
                                <span class="status-badge ${statusClass}">
                                    ${record.status === 'present' ? 'Tepat Waktu' : 
                                      record.status === 'late' ? 'Telat' : 'Absen'}
                                </span>
                            </td>
                            <td>${record.location || '-'}</td>
                        `;
                        
                        tableBody.appendChild(row);
                    });
                } else {
                    document.getElementById('attendanceTable').style.display = 'none';
                    document.getElementById('noAttendance').style.display = 'block';
                }
            } catch (error) {
                console.error('Error:', error);
            }
        }
    </script>
